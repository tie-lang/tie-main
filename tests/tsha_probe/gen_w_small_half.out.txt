// ============================================================================
// W=3 单环特化内核（n=12/16 r 半宽；L=W 单环）——语义与通用 ring_mix 单环逐式一致。
// ============================================================================
struct Lanes3h {
    var M0: i64 = 0
    var N0: i64 = 0
    var M1: i64 = 0
    var N1: i64 = 0
    var M2: i64 = 0
    var N2: i64 = 0
}

func ring3h_mix(l: Lanes3h, r: i64
            , skey: i64
            , M0: i64, N0: i64, M1: i64, N1: i64, mA: i64
            , rcon: table<i64>, rcon_idx: i64
            ) -> Lanes3h {
    var rA = (r * 3 + (skey & 7)) & 31
    var rB = (r * 7 + ((skey >> 3) & 7)) & 31
    var (q0m, q0n) = tadd2(l.M0, l.N0,
                    rrp16(l.M2, rA), rrp16(l.N2, rA))
    var (s0m, s0n) = tadd2(q0m, q0n,
                    rrp16(l.M0, (rB + 3) & 31), rrp16(l.N0, (rB + 3) & 31))
    var (q1m, q1n) = tadd2(l.M1, l.N1,
                    rrp16(l.M0, (rA + 5) & 31), rrp16(l.N0, (rA + 5) & 31))
    var (s1m, s1n) = tadd2(q1m, q1n,
                    rrp16(l.M1, (rB + 10) & 31), rrp16(l.N1, (rB + 10) & 31))
    var (q2m, q2n) = tadd2(l.M2, l.N2,
                    rrp16(l.M1, (rA + 10) & 31), rrp16(l.N1, (rA + 10) & 31))
    var (s2m, s2n) = tadd2(q2m, q2n,
                    rrp16(l.M2, (rB + 17) & 31), rrp16(l.N2, (rB + 17) & 31))
    l.M0 = s0m
    l.N0 = s0n
    l.M1 = s1m
    l.N1 = s1n
    l.M2 = s2m
    l.N2 = s2n
    var (p0m, p0n) = tmul2(s0m, s0n, s2m, s2n)
    l.M1 = (l.M1 ^ p0m) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ p0n) & 0xFFFFFFFF
    var (p1m, p1n) = tmul2(s1m, s1n, s0m, s0n)
    l.M2 = (l.M2 ^ p1m) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ p1n) & 0xFFFFFFFF
    var (p2m, p2n) = tmul2(s2m, s2n, s1m, s1n)
    l.M0 = (l.M0 ^ p2m) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ p2n) & 0xFFFFFFFF
    var maj = quant3(s0m, s0n, s1m, s1n, s2m, s2n)
    l.M2 = (l.M2 ^ maj) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ rrp16(maj, (rB + 7) & 31)) & 0xFFFFFFFF
    var (im, in2) = tadd2(l.M0, l.N0, rrp16(M0, mA), rrp16(N0, mA))
    l.M0 = im
    l.N0 = in2
    var (jm, jn2) = tadd2(l.M1, l.N1, rrp16(M1, (mA + 7) & 31), rrp16(N1, (mA + 7) & 31))
    l.M1 = jm
    l.N1 = jn2
    l.M0 = (l.M0 ^ rcon[rcon_idx & 15]) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ rrp16(rcon[(rcon_idx + 1) & 15], (r * 5) & 31)) & 0xFFFFFFFF
    return l
}

func ring3h_mixF(l: Lanes3h, r: i64
            , skey: i64
            ) -> Lanes3h {
    var rA = (r * 3 + (skey & 7)) & 31
    var rB = (r * 7 + ((skey >> 3) & 7)) & 31
    var (q0m, q0n) = tadd2(l.M0, l.N0,
                    rrp16(l.M2, rA), rrp16(l.N2, rA))

                    rrp16(l.M0, (rA + 5) & 31), rrp16(l.N0, (rA + 5) & 31))
    var (s1m, s1n) = tadd2(q1m, q1n,
                    rrp16(l.M1, (rB + 10) & 31), rrp16(l.N1, (rB + 10) & 31))
    var (q2m, q2n) = tadd2(l.M2, l.N2,
                    rrp16(l.M1, (rA + 10) & 31), rrp16(l.N1, (rA + 10) & 31))
    var (s2m, s2n) = tadd2(q2m, q2n,
                    rrp16(l.M2, (rB + 17) & 31), rrp16(l.N2, (rB + 17) & 31))
    l.M0 = s0m
    l.N0 = s0n
    l.M1 = s1m

    l.N2 = s2n
    var (p0m, p0n) = tmul2(s0m, s0n, s2m, s2n)
    l.M1 = (l.M1 ^ p0m) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ p0n) & 0xFFFFFFFF
    var (p1m, p1n) = tmul2(s1m, s1n, s0m, s0n)
    l.M2 = (l.M2 ^ p1m) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ p1n) & 0xFFFFFFFF
    var (p2m, p2n) = tmul2(s2m, s2n, s1m, s1n)
    l.M0 = (l.M0 ^ p2m) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ p2n) & 0xFFFFFFFF
    var maj = quant3(s0m, s0n, s1m, s1n, s2m, s2n)
    l.M2 = (l.M2 ^ maj) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ rrp16(maj, (rB + 7) & 31)) & 0xFFFFFFFF
    return l
}

func absorb3h(l: Lanes3h, Pw: Planes16, skey: i64) -> Lanes3h {
    var w: i64 = 0
    while w < 16 {
        var pv: i64 = 0
        if w == 0 {
            pv = Pw.w0
        } else if w == 1 {
            pv = Pw.w1
        } else if w == 2 {
            pv = Pw.w2
        } else if w == 3 {
            pv = Pw.w3
        } else if w == 4 {
            pv = Pw.w4
        } else if w == 5 {
            pv = Pw.w5
        } else if w == 6 {
            pv = Pw.w6
        } else if w == 7 {
            pv = Pw.w7
        } else if w == 8 {
            pv = Pw.w8
        } else if w == 9 {
            pv = Pw.w9
        } else if w == 10 {
            pv = Pw.w10
        } else if w == 11 {
            pv = Pw.w11
        } else if w == 12 {
            pv = Pw.w12
        } else if w == 13 {
            pv = Pw.w13
        } else if w == 14 {
            pv = Pw.w14
        } else {
            pv = Pw.w15
        }
        if pv != 0 {
            var a = (w * 5 + ((skey >> (w & 7)) & 7)) & 31
            var an = w % 3
            var v = rrp16(pv, a)
            var v2 = rrp16(pv, a + 17)
            if an == 0 {
                var (t0m, t0n) = tadd2(l.M0, l.N0, v, v2)
                l.M0 = t0m & 0xFFFFFFFF
                l.N0 = t0n & 0xFFFFFFFF
            } else if an == 1 {
                var (t1m, t1n) = tadd2(l.M1, l.N1, v, v2)
                l.M1 = t1m & 0xFFFFFFFF
                l.N1 = t1n & 0xFFFFFFFF
            } else {
                var (t2m, t2n) = tadd2(l.M2, l.N2, v, v2)
                l.M2 = t2m & 0xFFFFFFFF
                l.N2 = t2n & 0xFFFFFFFF
            }
        }
        w = w + 1
    }
    return l
}

// W=3 r 半宽终筛
func fin_synth3h(h: ref table<i64>) {
    var l = Lanes3h()
    l.M0 = h[0] & 0xFFFFFFFF
    l.N0 = rrp16(h[1], 7)
    l.M1 = h[1] & 0xFFFFFFFF
    l.N1 = rrp16(h[2], 7)
    l.M2 = h[2] & 0xFFFFFFFF
    l.N2 = rrp16(h[0], 7)
    var sf: i64 = 0
    while sf < 4 {
        l = ring3h_mixF(l, sf, 0x13579BDF)
        sf = sf + 1
    }
    h[0] = (l.M0 ^ rrp16(l.N0, 0)) & 0xFFFFFFFF
    h[1] = (l.M1 ^ rrp16(l.N1, 5)) & 0xFFFFFFFF
    h[2] = (l.M2 ^ rrp16(l.N2, 10)) & 0xFFFFFFFF
}

// W=3 r 半宽单块压缩（r 半宽 R=8，R 各档）
pub func compress_w3h(h: ref table<i64>, msg: string, pos: i64, nbytes: i64,
            t_lo: i64, t_hi: i64, last: bool,
            iv: table<i64>, rcon: table<i64>, R: i64) {
    var (M0, N0, M1, N1, Pw, skey) = w48_planes_skey(msg, pos, nbytes)
    var l = Lanes3h()
    l.M0 = h[0] & 0xFFFFFFFF
    l.N0 = rrp16(h[1], 7)
    l.M1 = h[1] & 0xFFFFFFFF
    l.N1 = rrp16(h[2], 7)
    l.M2 = h[2] & 0xFFFFFFFF
    l.N2 = rrp16(h[0], 7)
    l.M0 = (l.M0 ^ (iv[0] & 0xFFFF)) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ (iv[4] & 0xFFFF)) & 0xFFFFFFFF
    l.M1 = (l.M1 ^ (iv[1] & 0xFFFF)) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ (iv[5] & 0xFFFF)) & 0xFFFFFFFF
    l.M2 = (l.M2 ^ (iv[2] & 0xFFFF)) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ (iv[6] & 0xFFFF)) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ (t_lo & 0xFFFF)) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ (t_hi & 0xFFFF)) & 0xFFFFFFFF
    if last {
        l.N1 = (l.N1 ^ 0xFFFFFFFF) & 0xFFFFFFFF
    }
    l = absorb3h(l, Pw, skey)
    var mA: i64 = 0
    var r: i64 = 0
    while r < R {
        mA = (r * 5 + ((skey >> 6) & 7)) & 31
        l = ring3h_mix(l, r, skey, M0, N0, M1, N1, mA, rcon, r)
        r = r + 1
    }
    h[0] = (h[0] ^ l.M0 ^ rrp16(l.N0, 0)) & 0xFFFFFFFF
    h[1] = (h[1] ^ l.M1 ^ rrp16(l.N1, 3)) & 0xFFFFFFFF
    h[2] = (h[2] ^ l.M2 ^ rrp16(l.N2, 6)) & 0xFFFFFFFF
    fin_synth3h(h)
}

            t_lo: i64, t_hi: i64, last: bool,
            iv: table<i64>, rcon: table<i64>, R: i64) {

    var l = Lanes4h()

    l.N0 = rrp16(h[2], 7)
    l.M1 = h[1] & 0xFFFFFFFF
    l.N1 = rrp16(h[3], 7)
    l.M2 = h[2] & 0xFFFFFFFF
    l.N2 = rrp16(h[0], 7)
    l.M3 = h[3] & 0xFFFFFFFF
    l.N3 = rrp16(h[1], 7)
    l.M0 = (l.M0 ^ (iv[0] & 0xFFFF)) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ (iv[4] & 0xFFFF)) & 0xFFFFFFFF
    l.M1 = (l.M1 ^ (iv[1] & 0xFFFF)) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ (iv[5] & 0xFFFF)) & 0xFFFFFFFF

    l.N3 = (l.N3 ^ (iv[7] & 0xFFFF)) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ (t_lo & 0xFFFF)) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ (t_hi & 0xFFFF)) & 0xFFFFFFFF
    if last {
        l.N2 = (l.N2 ^ 0xFFFFFFFF) & 0xFFFFFFFF
    }
    l = absorb4h(l, Pw, skey)
    var mA: i64 = 0
    var r: i64 = 0
    while r < R {
        mA = (r * 5 + ((skey >> 6) & 7)) & 31
        l = ring4h_mix(l, r, skey, M0, N0, M1, N1, mA, rcon, r)
        r = r + 1
    }
    h[0] = (h[0] ^ l.M0 ^ rrp16(l.N0, 0)) & 0xFFFFFFFF
    h[1] = (h[1] ^ l.M1 ^ rrp16(l.N1, 3)) & 0xFFFFFFFF
    h[2] = (h[2] ^ l.M2 ^ rrp16(l.N2, 6)) & 0xFFFFFFFF
    h[3] = (h[3] ^ l.M3 ^ rrp16(l.N3, 9)) & 0xFFFFFFFF
    fin_synth4h(h)
}

// ============================================================================
// W=6 单环特化内核（n=32 r 半宽；L=W 单环）——语义与通用 ring_mix 单环逐式一致。
// ============================================================================
struct Lanes6h {
    var M0: i64 = 0
    var N0: i64 = 0
    var M1: i64 = 0
    var N1: i64 = 0
    var M2: i64 = 0
    var N2: i64 = 0
    var M3: i64 = 0
    var N3: i64 = 0
    var M4: i64 = 0
    var N4: i64 = 0
    var M5: i64 = 0
    var N5: i64 = 0
}

func ring6h_mix(l: Lanes6h, r: i64
            , skey: i64
            , M0: i64, N0: i64, M1: i64, N1: i64, mA: i64
            , rcon: table<i64>, rcon_idx: i64
            ) -> Lanes6h {
    var rA = (r * 3 + (skey & 7)) & 31
    var rB = (r * 7 + ((skey >> 3) & 7)) & 31
    var (q0m, q0n) = tadd2(l.M0, l.N0,
                    rrp16(l.M2, rA), rrp16(l.N2, rA))
    var (s0m, s0n) = tadd2(q0m, q0n,
                    rrp16(l.M3, (rB + 3) & 31), rrp16(l.N3, (rB + 3) & 31))
    var (q1m, q1n) = tadd2(l.M1, l.N1,
                    rrp16(l.M3, (rA + 5) & 31), rrp16(l.N3, (rA + 5) & 31))
    var (s1m, s1n) = tadd2(q1m, q1n,
                    rrp16(l.M4, (rB + 10) & 31), rrp16(l.N4, (rB + 10) & 31))
    var (q2m, q2n) = tadd2(l.M2, l.N2,
                    rrp16(l.M4, (rA + 10) & 31), rrp16(l.N4, (rA + 10) & 31))
    var (s2m, s2n) = tadd2(q2m, q2n,
                    rrp16(l.M5, (rB + 17) & 31), rrp16(l.N5, (rB + 17) & 31))
    var (q3m, q3n) = tadd2(l.M3, l.N3,
                    rrp16(l.M5, (rA + 15) & 31), rrp16(l.N5, (rA + 15) & 31))
    var (s3m, s3n) = tadd2(q3m, q3n,
                    rrp16(l.M0, (rB + 24) & 31), rrp16(l.N0, (rB + 24) & 31))
    var (q4m, q4n) = tadd2(l.M4, l.N4,
                    rrp16(l.M0, (rA + 20) & 31), rrp16(l.N0, (rA + 20) & 31))
    var (s4m, s4n) = tadd2(q4m, q4n,
                    rrp16(l.M1, (rB + 31) & 31), rrp16(l.N1, (rB + 31) & 31))
    var (q5m, q5n) = tadd2(l.M5, l.N5,
                    rrp16(l.M1, (rA + 25) & 31), rrp16(l.N1, (rA + 25) & 31))
    var (s5m, s5n) = tadd2(q5m, q5n,
                    rrp16(l.M2, (rB + 6) & 31), rrp16(l.N2, (rB + 6) & 31))
    l.M0 = s0m
    l.N0 = s0n
    l.M1 = s1m
    l.N1 = s1n
    l.M2 = s2m
    l.N2 = s2n
    l.M3 = s3m
    l.N3 = s3n
    l.M4 = s4m
    l.N4 = s4n
    l.M5 = s5m
    l.N5 = s5n
    var (p0m, p0n) = tmul2(s0m, s0n, s2m, s2n)
    l.M1 = (l.M1 ^ p0m) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ p0n) & 0xFFFFFFFF
    var (p1m, p1n) = tmul2(s1m, s1n, s3m, s3n)
    l.M2 = (l.M2 ^ p1m) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ p1n) & 0xFFFFFFFF
    var (p2m, p2n) = tmul2(s2m, s2n, s4m, s4n)
    l.M3 = (l.M3 ^ p2m) & 0xFFFFFFFF
    l.N3 = (l.N3 ^ p2n) & 0xFFFFFFFF
    var (p3m, p3n) = tmul2(s3m, s3n, s5m, s5n)
    l.M4 = (l.M4 ^ p3m) & 0xFFFFFFFF
    l.N4 = (l.N4 ^ p3n) & 0xFFFFFFFF
    var (p4m, p4n) = tmul2(s4m, s4n, s0m, s0n)
    l.M5 = (l.M5 ^ p4m) & 0xFFFFFFFF
    l.N5 = (l.N5 ^ p4n) & 0xFFFFFFFF
    var (p5m, p5n) = tmul2(s5m, s5n, s1m, s1n)
    l.M0 = (l.M0 ^ p5m) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ p5n) & 0xFFFFFFFF
    var maj = quant3(s0m, s0n, s1m, s1n, s2m, s2n)
    l.M5 = (l.M5 ^ maj) & 0xFFFFFFFF
    l.N5 = (l.N5 ^ rrp16(maj, (rB + 7) & 31)) & 0xFFFFFFFF
    var (im, in2) = tadd2(l.M0, l.N0, rrp16(M0, mA), rrp16(N0, mA))
    l.M0 = im
    l.N0 = in2
    var (jm, jn2) = tadd2(l.M1, l.N1, rrp16(M1, (mA + 7) & 31), rrp16(N1, (mA + 7) & 31))
    l.M1 = jm
    l.N1 = jn2
    l.M0 = (l.M0 ^ rcon[rcon_idx & 15]) & 0xFFFFFFFF
    l.N5 = (l.N5 ^ rrp16(rcon[(rcon_idx + 1) & 15], (r * 5) & 31)) & 0xFFFFFFFF
    return l
}

func ring6h_mixF(l: Lanes6h, r: i64
            , skey: i64
            ) -> Lanes6h {
    var rA = (r * 3 + (skey & 7)) & 31
    var rB = (r * 7 + ((skey >> 3) & 7)) & 31
    var (q0m, q0n) = tadd2(l.M0, l.N0,
                    rrp16(l.M2, rA), rrp16(l.N2, rA))
    var (s0m, s0n) = tadd2(q0m, q0n,
                    rrp16(l.M3, (rB + 3) & 31), rrp16(l.N3, (rB + 3) & 31))
    var (q1m, q1n) = tadd2(l.M1, l.N1,
                    rrp16(l.M3, (rA + 5) & 31), rrp16(l.N3, (rA + 5) & 31))
    var (s1m, s1n) = tadd2(q1m, q1n,
                    rrp16(l.M4, (rB + 10) & 31), rrp16(l.N4, (rB + 10) & 31))
    var (q2m, q2n) = tadd2(l.M2, l.N2,
                    rrp16(l.M4, (rA + 10) & 31), rrp16(l.N4, (rA + 10) & 31))
    var (s2m, s2n) = tadd2(q2m, q2n,
                    rrp16(l.M5, (rB + 17) & 31), rrp16(l.N5, (rB + 17) & 31))
    var (q3m, q3n) = tadd2(l.M3, l.N3,
                    rrp16(l.M5, (rA + 15) & 31), rrp16(l.N5, (rA + 15) & 31))
    var (s3m, s3n) = tadd2(q3m, q3n,
                    rrp16(l.M0, (rB + 24) & 31), rrp16(l.N0, (rB + 24) & 31))
    var (q4m, q4n) = tadd2(l.M4, l.N4,
                    rrp16(l.M0, (rA + 20) & 31), rrp16(l.N0, (rA + 20) & 31))
    var (s4m, s4n) = tadd2(q4m, q4n,
                    rrp16(l.M1, (rB + 31) & 31), rrp16(l.N1, (rB + 31) & 31))
    var (q5m, q5n) = tadd2(l.M5, l.N5,
                    rrp16(l.M1, (rA + 25) & 31), rrp16(l.N1, (rA + 25) & 31))
    var (s5m, s5n) = tadd2(q5m, q5n,
                    rrp16(l.M2, (rB + 6) & 31), rrp16(l.N2, (rB + 6) & 31))
    l.M0 = s0m
    l.N0 = s0n
    l.M1 = s1m
    l.N1 = s1n
    l.M2 = s2m
    l.N2 = s2n
    l.M3 = s3m
    l.N3 = s3n
    l.M4 = s4m
    l.N4 = s4n
    l.M5 = s5m
    l.N5 = s5n
    var (p0m, p0n) = tmul2(s0m, s0n, s2m, s2n)
    l.M1 = (l.M1 ^ p0m) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ p0n) & 0xFFFFFFFF
    var (p1m, p1n) = tmul2(s1m, s1n, s3m, s3n)
    l.M2 = (l.M2 ^ p1m) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ p1n) & 0xFFFFFFFF
    var (p2m, p2n) = tmul2(s2m, s2n, s4m, s4n)
    l.M3 = (l.M3 ^ p2m) & 0xFFFFFFFF
    l.N3 = (l.N3 ^ p2n) & 0xFFFFFFFF
    var (p3m, p3n) = tmul2(s3m, s3n, s5m, s5n)
    l.M4 = (l.M4 ^ p3m) & 0xFFFFFFFF
    l.N4 = (l.N4 ^ p3n) & 0xFFFFFFFF
    var (p4m, p4n) = tmul2(s4m, s4n, s0m, s0n)
    l.M5 = (l.M5 ^ p4m) & 0xFFFFFFFF
    l.N5 = (l.N5 ^ p4n) & 0xFFFFFFFF
    var (p5m, p5n) = tmul2(s5m, s5n, s1m, s1n)
    l.M0 = (l.M0 ^ p5m) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ p5n) & 0xFFFFFFFF
    var maj = quant3(s0m, s0n, s1m, s1n, s2m, s2n)
    l.M5 = (l.M5 ^ maj) & 0xFFFFFFFF
    l.N5 = (l.N5 ^ rrp16(maj, (rB + 7) & 31)) & 0xFFFFFFFF
    return l
}

func absorb6h(l: Lanes6h, Pw: Planes16, skey: i64) -> Lanes6h {
    var w: i64 = 0
    while w < 16 {
        var pv: i64 = 0
        if w == 0 {
            pv = Pw.w0
        } else if w == 1 {
            pv = Pw.w1
        } else if w == 2 {
            pv = Pw.w2
        } else if w == 3 {
            pv = Pw.w3
        } else if w == 4 {
            pv = Pw.w4
        } else if w == 5 {
            pv = Pw.w5
        } else if w == 6 {
            pv = Pw.w6
        } else if w == 7 {
            pv = Pw.w7
        } else if w == 8 {
            pv = Pw.w8
        } else if w == 9 {
            pv = Pw.w9
        } else if w == 10 {
            pv = Pw.w10
        } else if w == 11 {
            pv = Pw.w11
        } else if w == 12 {
            pv = Pw.w12
        } else if w == 13 {
            pv = Pw.w13
        } else if w == 14 {
            pv = Pw.w14
        } else {
            pv = Pw.w15
        }
        if pv != 0 {
            var a = (w * 5 + ((skey >> (w & 7)) & 7)) & 31
            var an = w % 6
            var v = rrp16(pv, a)
            var v2 = rrp16(pv, a + 17)
            if an == 0 {
                var (t0m, t0n) = tadd2(l.M0, l.N0, v, v2)
                l.M0 = t0m & 0xFFFFFFFF
                l.N0 = t0n & 0xFFFFFFFF
            } else if an == 1 {
                var (t1m, t1n) = tadd2(l.M1, l.N1, v, v2)
                l.M1 = t1m & 0xFFFFFFFF
                l.N1 = t1n & 0xFFFFFFFF
            } else if an == 2 {
                var (t2m, t2n) = tadd2(l.M2, l.N2, v, v2)
                l.M2 = t2m & 0xFFFFFFFF
                l.N2 = t2n & 0xFFFFFFFF
            } else if an == 3 {
                var (t3m, t3n) = tadd2(l.M3, l.N3, v, v2)
                l.M3 = t3m & 0xFFFFFFFF
                l.N3 = t3n & 0xFFFFFFFF
            } else if an == 4 {
                var (t4m, t4n) = tadd2(l.M4, l.N4, v, v2)
                l.M4 = t4m & 0xFFFFFFFF
                l.N4 = t4n & 0xFFFFFFFF
            } else {
                var (t5m, t5n) = tadd2(l.M5, l.N5, v, v2)
                l.M5 = t5m & 0xFFFFFFFF
                l.N5 = t5n & 0xFFFFFFFF
            }
        }
        w = w + 1
    }
    return l
}

// W=6 r 半宽终筛
func fin_synth6h(h: ref table<i64>) {
    var l = Lanes6h()
    l.M0 = h[0] & 0xFFFFFFFF
    l.N0 = rrp16(h[1], 7)
    l.M1 = h[1] & 0xFFFFFFFF
    l.N1 = rrp16(h[2], 7)
    l.M2 = h[2] & 0xFFFFFFFF
    l.N2 = rrp16(h[3], 7)
    l.M3 = h[3] & 0xFFFFFFFF
    l.N3 = rrp16(h[4], 7)
    l.M4 = h[4] & 0xFFFFFFFF
    l.N4 = rrp16(h[5], 7)
    l.M5 = h[5] & 0xFFFFFFFF
    l.N5 = rrp16(h[0], 7)
    var sf: i64 = 0
    while sf < 4 {
        l = ring6h_mixF(l, sf, 0x13579BDF)
        sf = sf + 1
    }
    h[0] = (l.M0 ^ rrp16(l.N0, 0)) & 0xFFFFFFFF
    h[1] = (l.M1 ^ rrp16(l.N1, 5)) & 0xFFFFFFFF
    h[2] = (l.M2 ^ rrp16(l.N2, 10)) & 0xFFFFFFFF
    h[3] = (l.M3 ^ rrp16(l.N3, 15)) & 0xFFFFFFFF
    h[4] = (l.M4 ^ rrp16(l.N4, 20)) & 0xFFFFFFFF
    h[5] = (l.M5 ^ rrp16(l.N5, 25)) & 0xFFFFFFFF
}

// W=6 r 半宽单块压缩（r 半宽 R=8，R 各档）
pub func compress_w6h(h: ref table<i64>, msg: string, pos: i64, nbytes: i64,
            t_lo: i64, t_hi: i64, last: bool,
            iv: table<i64>, rcon: table<i64>, R: i64) {
    var (M0, N0, M1, N1, Pw, skey) = w48_planes_skey(msg, pos, nbytes)
    var l = Lanes6h()
    l.M0 = h[0] & 0xFFFFFFFF
    l.N0 = rrp16(h[3], 7)
    l.M1 = h[1] & 0xFFFFFFFF
    l.N1 = rrp16(h[4], 7)
    l.M2 = h[2] & 0xFFFFFFFF
    l.N2 = rrp16(h[5], 7)
    l.M3 = h[3] & 0xFFFFFFFF
    l.N3 = rrp16(h[0], 7)
    l.M4 = h[4] & 0xFFFFFFFF
    l.N4 = rrp16(h[1], 7)
    l.M5 = h[5] & 0xFFFFFFFF
    l.N5 = rrp16(h[2], 7)
    l.M0 = (l.M0 ^ (iv[0] & 0xFFFF)) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ (iv[4] & 0xFFFF)) & 0xFFFFFFFF
    l.M1 = (l.M1 ^ (iv[1] & 0xFFFF)) & 0xFFFFFFFF
    l.N1 = (l.N1 ^ (iv[5] & 0xFFFF)) & 0xFFFFFFFF
    l.M2 = (l.M2 ^ (iv[2] & 0xFFFF)) & 0xFFFFFFFF
    l.N2 = (l.N2 ^ (iv[6] & 0xFFFF)) & 0xFFFFFFFF
    l.M3 = (l.M3 ^ (iv[3] & 0xFFFF)) & 0xFFFFFFFF
    l.N3 = (l.N3 ^ (iv[7] & 0xFFFF)) & 0xFFFFFFFF
    l.M4 = (l.M4 ^ (iv[4] & 0xFFFF)) & 0xFFFFFFFF
    l.N4 = (l.N4 ^ (iv[0] & 0xFFFF)) & 0xFFFFFFFF
    l.M5 = (l.M5 ^ (iv[5] & 0xFFFF)) & 0xFFFFFFFF
    l.N5 = (l.N5 ^ (iv[1] & 0xFFFF)) & 0xFFFFFFFF
    l.N0 = (l.N0 ^ (t_lo & 0xFFFF)) & 0xFFFFFFFF
    l.N3 = (l.N3 ^ (t_hi & 0xFFFF)) & 0xFFFFFFFF
    if last {
        l.N3 = (l.N3 ^ 0xFFFFFFFF) & 0xFFFFFFFF
    }
    l = absorb6h(l, Pw, skey)
    var mA: i64 = 0
    var r: i64 = 0
    while r < R {
        mA = (r * 5 + ((skey >> 6) & 7)) & 31
        l = ring6h_mix(l, r, skey, M0, N0, M1, N1, mA, rcon, r)
        r = r + 1
    }
    h[0] = (h[0] ^ l.M0 ^ rrp16(l.N0, 0)) & 0xFFFFFFFF
    h[1] = (h[1] ^ l.M1 ^ rrp16(l.N1, 3)) & 0xFFFFFFFF
    h[2] = (h[2] ^ l.M2 ^ rrp16(l.N2, 6)) & 0xFFFFFFFF
    h[3] = (h[3] ^ l.M3 ^ rrp16(l.N3, 9)) & 0xFFFFFFFF
    h[4] = (h[4] ^ l.M4 ^ rrp16(l.N4, 12)) & 0xFFFFFFFF
    h[5] = (h[5] ^ l.M5 ^ rrp16(l.N5, 15)) & 0xFFFFFFFF
    fin_synth6h(h)
}
